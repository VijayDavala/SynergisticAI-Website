STEP 3 — SUPABASE DATABASE SETUP
──────────────────────────────────
Open your Supabase project → SQL Editor → run these in order:

-- Enable pgvector
CREATE EXTENSION IF NOT EXISTS vector;

-- Sessions table
CREATE TABLE sessions (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  visitor_id text NOT NULL,
  started_at timestamptz NOT NULL DEFAULT now(),
  last_activity timestamptz DEFAULT now(),
  page_url text,
  utm_source text, utm_medium text, utm_campaign text,
  is_returning boolean NOT NULL DEFAULT false,
  status text NOT NULL DEFAULT 'active'
);
CREATE INDEX ON sessions (visitor_id);

-- Conversations table
CREATE TABLE conversations (
  id bigserial PRIMARY KEY,
  session_id uuid REFERENCES sessions(id),
  role text NOT NULL,
  content text NOT NULL,
  intent text,
  confidence float4,
  tokens_used integer,
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ON conversations (session_id);

-- Lead qualifications (extends your existing leads table)
CREATE TABLE lead_qualifications (
  id bigserial PRIMARY KEY,
  lead_id bigint REFERENCES leads(id),
  session_id uuid REFERENCES sessions(id),
  company_website text,
  industry text,
  employee_count text,
  monthly_budget text,
  business_challenge text,
  decision_maker boolean,
  urgency text,
  lead_score integer DEFAULT 0,
  lead_category text DEFAULT 'cold',
  hubspot_contact_id text,
  hubspot_deal_id text,
  conversation_summary text,
  created_at timestamptz NOT NULL DEFAULT now()
);

-- Appointments table
CREATE TABLE appointments (
  id bigserial PRIMARY KEY,
  session_id uuid REFERENCES sessions(id),
  lead_id bigint REFERENCES leads(id),
  calendly_event_id text,
  scheduled_at timestamptz,
  duration_mins integer DEFAULT 30,
  status text NOT NULL DEFAULT 'scheduling',
  meeting_link text,
  created_at timestamptz NOT NULL DEFAULT now()
);

-- Follow-ups table
CREATE TABLE follow_ups (
  id bigserial PRIMARY KEY,
  lead_id bigint NOT NULL REFERENCES leads(id),
  sequence_day integer NOT NULL,
  scheduled_at timestamptz NOT NULL,
  sent_at timestamptz,
  email_subject text,
  status text NOT NULL DEFAULT 'pending',
  open_count integer DEFAULT 0,
  click_count integer DEFAULT 0
);
CREATE INDEX ON follow_ups (status, scheduled_at);

-- Knowledge base (for RAG)
CREATE TABLE knowledge_base_docs (
  id bigserial PRIMARY KEY,
  title text NOT NULL,
  source_type text NOT NULL,
  source_url text,
  content text NOT NULL,
  embedding vector(1536),
  chunk_index integer NOT NULL DEFAULT 0,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz
);

-- Similarity search function
CREATE OR REPLACE FUNCTION match_documents(
  query_embedding vector(1536),
  match_threshold float DEFAULT 0.75,
  match_count int DEFAULT 5
) RETURNS TABLE (id bigint, title text, content text, similarity float)
LANGUAGE SQL STABLE AS $$
  SELECT id, title, content,
    1 - (embedding <=> query_embedding) AS similarity
  FROM knowledge_base_docs
  WHERE 1 - (embedding <=> query_embedding) > match_threshold
  ORDER BY embedding <=> query_embedding
  LIMIT match_count;
$$;

-- Vector index
CREATE INDEX ON knowledge_base_docs
  USING ivfflat (embedding vector_cosine_ops) WITH (lists = 100);
